Telegram-бот, который помогает создавать настоящие эмоциональные события в отношениях.

Проект ориентирован не только на отправку красивых сообщений, а на создание совместных воспоминаний, укрепление отношений и помощь людям на разных этапах общения — от знакомства до долгих отношений.

```bash
pip install -r requirements.txt
```

На основе `.env.example`.

```env
BOT_TOKEN=your_bot_token

```

```bash
python main.py
```

def should_skip_dir(name: str, include_hidden: bool) -> bool:
    if name in IGNORE_DIRS:
        return True

    if name.startswith(".") and not include_hidden:
        return True

    return False


def is_probably_binary(path: Path) -> bool:
    try:
        with path.open("rb") as f:
            chunk = f.read(8192)
            return b"\0" in chunk
    except OSError:
        return True


def read_text_file(path: Path):
    if is_probably_binary(path):
        return None

    # Сначала пробуем UTF-8, потом Windows-1251 на случай кириллицы
    for encoding in ("utf-8", "cp1251"):
        try:
            return path.read_text(encoding=encoding)
        except UnicodeDecodeError:
            continue
        except OSError as e:
            return f"[ERROR] Не удалось прочитать файл: {e}"

    # Если кодировка не определилась, читаем с заменой символов
    try:
        text = path.read_text(encoding="utf-8", errors="replace")
        return text + "\n[WARNING] Файл прочитан с заменой символов. Возможна нестандартная кодировка."
    except OSError as e:
        return f"[ERROR] Не удалось прочитать файл: {e}"


def main():
    parser = argparse.ArgumentParser(
        description="Собирает текстовые файлы проекта в один файл."
    )

    parser.add_argument(
        "source",
        help="Папка проекта, например: . или C:/projects/my_project",
    )

    parser.add_argument(
        "output",
        help="Файл, куда всё записать, например: all_project.txt",
    )

    parser.add_argument(
        "--max-size",
        type=int,
        default=2_000_000,
        help="Максимальный размер одного файла в байтах. По умолчанию 2 MB.",
    )

    parser.add_argument(
        "--include-hidden",
        action="store_true",
        help="Не пропускать скрытые файлы и папки, начинающиеся с точки.",
    )

    args = parser.parse_args()

    source = Path(args.source).expanduser().resolve()
    output = Path(args.output).expanduser().resolve()

    if not source.is_dir():
        raise SystemExit(f"Папка проекта не найдена: {source}")

    files = []

    for root, dirs, names in os.walk(source):
        root_path = Path(root)

        # Сортируем папки, чтобы порядок был предсказуемым
        dirs[:] = sorted(
            d for d in dirs if not should_skip_dir(d, args.include_hidden)
        )

        for name in sorted(names):
            path = root_path / name

            try:
                resolved_path = path.resolve()
            except OSError:
                continue

            # Пропускаем сам итоговый файл, если он лежит внутри проекта
            if resolved_path == output:
                continue

            # Пропускаем скрытые файлы, если не указан --include-hidden
            if not args.include_hidden and name.startswith("."):
                continue

            # Пропускаем потенциально секретные файлы
            if name in IGNORE_FILES:
                continue

            # Пропускаем нежелательные расширения
            if path.suffix.lower() in IGNORE_SUFFIXES:
                continue

            if path.is_file():
                files.append(path)

    output.parent.mkdir(parents=True, exist_ok=True)

    with output.open("w", encoding="utf-8") as out:
        out.write(f"PROJECT DUMP: {source}\n")

        for path in files:
            relative_path = path.relative_to(source).as_posix()

            out.write("\n\n")
            out.write("=" * 20)
            out.write(f" FILE: {relative_path} ")
            out.write("=" * 20)
            out.write("\n")

            try:
                size = path.stat().st_size
            except OSError as e:
                out.write(f"[ERROR] Не удалось получить информацию о файле: {e}\n")
                continue

            if size > args.max_size:
                out.write(
                    f"[SKIPPED] Файл больше лимита: {size} байт. "
                    f"Лимит: {args.max_size} байт.\n"
                )
                continue

            content = read_text_file(path)

            if content is None:
                out.write("[SKIPPED] Похоже, это двоичный файл.\n")
            else:
                out.write(content.rstrip())
                out.write("\n")

    print(f"Готово: {output}")
    print(f"Файлов обработано: {len(files)}")


if __name__ == "__main__":
    main()

import asyncio
import logging
import sys
from datetime import datetime, timezone

from aiogram import Bot, Dispatcher
from aiogram.fsm.storage.redis import RedisStorage
from aiogram.types import BotCommand
from apscheduler.schedulers.asyncio import AsyncIOScheduler
from sqlalchemy import text

from bot.handlers import router
from bot.middlewares.db import DbSessionMiddleware
from config import settings
from database import async_session_maker, engine
from database.repositories.letter import LetterRepository
from database.repositories.user import UserRepository
from services.letter_service import LetterService

logging.basicConfig(
    level=logging.INFO,
    stream=sys.stdout,
)


async def check_db_connection() -> None:
    """Проверяет подключение к базе данных через запрос SELECT 1."""
    try:
        async with engine.connect() as conn:
            await conn.execute(text("SELECT 1"))
        logging.info("Успешное подключение к PostgreSQL!")
    except Exception as e:
        logging.critical(f"Ошибка при подключении к базе данных: {e}")
        raise e


async def set_bot_commands(bot: Bot) -> None:
    """Устанавливает команды бота для автодополнения в поле ввода."""
    commands = [
        BotCommand(command="start", description="Начать работу с ботом"),
        BotCommand(command="profile", description="Мой профиль"),
        BotCommand(command="couple", description="Управление парой"),
        BotCommand(command="letters", description="Мои активные письма"),
        BotCommand(command="settings", description="Настройки"),
        BotCommand(command="leave", description="Выйти из пары"),
        BotCommand(command="cancel", description="Отменить текущее действие"),
    ]
    await bot.set_my_commands(commands)


async def check_pending_letters(bot: Bot) -> None:
    """Фоновая задача: проверка и доставка писем каждую минуту."""
    async with async_session_maker() as session:
        letter_repo = LetterRepository(session)
        user_repo = UserRepository(session)
        letter_service = LetterService(letter_repo, user_repo, bot)
        
        now = datetime.now(timezone.utc)
        
        # <-- ИЗМЕНЕНО: Оборачиваем в транзакцию для защиты от гонки
        async with session.begin():
            # get_pending теперь использует FOR UPDATE SKIP LOCKED
            pending_letters = await letter_repo.get_pending(now)
            
            for letter in pending_letters:
                # <-- ИЗМЕНЕНО: Идемпотентность (дополнительная защита)
                if letter.status != "pending":
                    continue
                
                try:
                    await letter_service.deliver_letter(letter)
                    logging.info(f"Письмо {letter.id} успешно доставлено.")
                except Exception as e:
                    logging.error(f"Ошибка при доставке письма {letter.id}: {e}")
                    # Пробрасываем исключение, чтобы session.begin() автоматически сделал rollback
                    raise


async def main() -> None:
    await check_db_connection()

    storage = RedisStorage.from_url(settings.redis_url)
    
    dp = Dispatcher(storage=storage)
    dp.update.middleware(DbSessionMiddleware(session_pool=async_session_maker))
    dp.include_router(router)

    # --- ИНИЦИАЛИЗАЦИЯ ПЛАНИРОВЩИКА ---
    scheduler = AsyncIOScheduler(timezone="UTC")
    
    async with Bot(token=settings.BOT_TOKEN) as bot:
        scheduler.add_job(
            check_pending_letters, 
            "interval", 
            minutes=1, 
            kwargs={"bot": bot},
            id="check_pending_letters_job",
            replace_existing=True
        )
        scheduler.start()
        logging.info("Планировщик задач запущен (проверка писем каждую минуту).")

        try:
            await set_bot_commands(bot)
            logging.info("Запуск бота...")
            await dp.start_polling(bot)
        finally:
            logging.info("Остановка бота: закрываем пул соединений БД и планировщик...")
            scheduler.shutdown()
            await engine.dispose()


if __name__ == "__main__":
    asyncio.run(main())

from config import settings

# Это объект конфигурации Alembic
config = context.config

# Настройка логирования из alembic.ini

# Динамически подставляем DSN из нашего config.py (избегаем дублирования в alembic.ini)

# Связываем Alembic с нашими ORM-моделями для автогенерации
target_metadata = Base.metadata


def run_migrations_offline() -> None:
    """Запуск миграций в 'offline' режиме (без живого подключения к БД)."""
    url = config.get_main_option("sqlalchemy.url")
    context.configure(
        url=url,
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
    )

    with context.begin_transaction():
        context.run_migrations()


def do_run_migrations(connection: Connection) -> None:
    context.configure(connection=connection, target_metadata=target_metadata)

    with context.begin_transaction():
        context.run_migrations()


async def run_migrations_online() -> None:
    """Запуск миграций в 'online' режиме (с подключением к asyncpg)."""
    connectable = async_engine_from_config(
        config.get_section(config.config_ini_section, {}),
        prefix="sqlalchemy.",

    action: Literal[

@router.message(Command("couple"))
async def cmd_couple(
    message: Message,
    user_service: UserService,
    couple_service: CoupleService,
) -> None:
    user_id = message.from_user.id
    couple = await couple_service.get_couple(user_id)
    
    if couple:
        partner_id = couple.user2_id if couple.user1_id == user_id else couple.user1_id
        partner = await user_service.get_profile(partner_id)
        partner_name = (partner.first_name or partner.username or "твоим партнёром") if partner else "твоим партнёром"

        )
        return
    
    await message.answer(
        "Ты сейчас не в паре. Что хочешь сделать?",
        reply_markup=get_couple_main_kb(),
        parse_mode="HTML"
    )

@router.message(CoupleStates.waiting_for_code, F.text)
async def process_join_code(
    message: Message,
    state: FSMContext,
    couple_service: CoupleService,
    user_service: UserService,
) -> None:
    if message.text.startswith('/'):
        return
    
    user_id = message.from_user.id
    code = message.text.strip().upper()

    try:
        couple = await couple_service.join_couple(user_id, code)
    except ValueError as e:
        if "уже в паре" in str(e):
            await state.clear()
        await message.answer(str(e), parse_mode="HTML")
        return

    await state.clear()
    
    user2 = await user_service.get_profile(user_id)
    user1 = await user_service.get_profile(couple.user1_id)
    
    user2_name = user2.first_name or user2.username or "твоим партнёром"
    user1_name = user1.first_name or user1.username or "твоим партнёром"
    
    await message.answer(
        f"❤️ Вы теперь пара с {user1_name}!",
        parse_mode="HTML"
    )
    
    await message.bot.send_message(
        chat_id=couple.user1_id,
        text=f"❤️ {user2_name} присоединился к тебе! Вы теперь пара!",
        parse_mode="HTML"
    )

@router.callback_query(LetterDeliveryAction.filter(F.action == "thanks"))
async def cb_delivery_thanks(
    callback: CallbackQuery,
    callback_data: LetterDeliveryAction,
    user_service: UserService,
    letter_service: LetterService,
) -> None:
    """Обработка кнопки [❤️ Спасибо]"""
    letter_id = callback_data.letter_id
    user_id = callback.from_user.id
    
    letter = await letter_service.get_letter_by_id(letter_id)
    if not letter:
        await callback.answer("Письмо не найдено", show_alert=True)
        return
    
    user = await user_service.get_profile(telegram_id=user_id)
    user_name = (user.first_name or user.username or "Твой партнер") if user else "Твой партнер"
    
    try:
        await callback.message.bot.send_message(
            chat_id=letter.sender_id,
            text=f"❤️ {user_name} получил(а) твоё письмо и отправил(а) ❤️",
            parse_mode="HTML"
        )
    except TelegramBadRequest as e:
        logger.warning(f"Не удалось уведомить отправителя {letter.sender_id}: {e}")
    
    await callback.message.edit_reply_markup(reply_markup=None)
    await callback.answer("❤️ Спасибо отправлено!")


@router.callback_query(LetterDeliveryAction.filter(F.action == "reply"))
async def cb_delivery_reply(
    callback: CallbackQuery,
    state: FSMContext,
    couple_service: CoupleService,
) -> None:
    """Обработка кнопки [💌 Ответить] — запуск FSM создания письма"""
    user_id = callback.from_user.id
    
    couple = await couple_service.get_couple(user_id)

        await callback.answer("Сначала создай пару 💑", show_alert=True)
        return

        "Выбери, когда твоё письмо будет открыто:",
        reply_markup=get_letter_date_keyboard(),
        parse_mode="HTML"
    )
    await callback.answer()

@router.callback_query(LetterAction.filter(F.action == "select_date"), LetterStates.choosing_date)
async def cb_select_date(
    callback: CallbackQuery,
    state: FSMContext,
    user_service: UserService,
) -> None:
    callback_data = LetterAction.unpack(callback.data)
    days = callback_data.days
    
    user = await user_service.get_profile(telegram_id=callback.from_user.id)
    user_tz = user.timezone if user else "UTC"
    
    if days is None:
        await state.set_state(LetterStates.entering_date)
        await callback.message.edit_text(
            "🗓 Введи <b>дату</b> открытия в формате <b>ДД.ММ.ГГГГ</b>\n\n"
            "Например: 01.01.2026",
            reply_markup=get_cancel_fsm_kb(),
            parse_mode="HTML"
        )
        await callback.answer()
        return
    
    now_local = to_local(datetime.now(timezone.utc), user_tz)
    open_date_local = now_local + timedelta(days=days)
    open_date_utc = to_utc(open_date_local, user_tz)
    
    await state.update_data(open_date=open_date_utc.isoformat())
    
    await state.set_state(LetterStates.entering_text)
    await callback.message.edit_text(
        f"📅 Письмо будет открыто через {days} дней "
        f"({open_date_local.strftime('%d.%m.%Y в %H:%M')})\n\n"
        "Теперь напиши текст письма:",
        reply_markup=get_cancel_fsm_kb(),
        parse_mode="HTML"
    )
    await callback.answer()


@router.message(LetterStates.entering_date, F.text.regexp(r"^\d{2}\.\d{2}\.\d{4}$"))
async def process_entering_date(message: Message, state: FSMContext) -> None:
    try:
        datetime.strptime(message.text.strip(), "%d.%m.%Y")
        await state.update_data(date_str=message.text.strip())
        await state.set_state(LetterStates.entering_time)
        await message.answer(
            "⏰ Теперь введи <b>время</b> в формате <b>ЧЧ:ММ</b>\n\n"
            "<i>Формат 24 часа.</i>\n"
            "Например: 12:00",
            reply_markup=get_cancel_fsm_kb(),
            parse_mode="HTML"
        )
    except ValueError:
        await message.answer(
            "Неверная дата. Используй формат <b>ДД.ММ.ГГГГ</b>", 
            reply_markup=get_cancel_fsm_kb(),
            parse_mode="HTML"
        )

@router.message(LetterStates.entering_time, F.text.regexp(r"^\d{2}:\d{2}$"))
async def process_entering_time(
    message: Message, 
    state: FSMContext,
    user_service: UserService,
) -> None:
    data = await state.get_data()
    date_str = data.get("date_str")
    time_str = message.text.strip()
    
    user = await user_service.get_profile(telegram_id=message.from_user.id)
    user_tz = user.timezone if user else "UTC"
    
    try:
        dt_str = f"{date_str} {time_str}"
        date_obj = datetime.strptime(dt_str, "%d.%m.%Y %H:%M")
        open_date_utc = to_utc(date_obj, user_tz)
        
        if open_date_utc <= datetime.now(timezone.utc):
            await message.answer(
                "⏳ Дата и время должны быть в будущем. Попробуй указать время позже.",
                reply_markup=get_cancel_fsm_kb()
            )
            return
        
        await state.update_data(open_date=open_date_utc.isoformat())
        await state.set_state(LetterStates.entering_text)
        
        open_date_local = to_local(open_date_utc, user_tz)
        
        await message.answer(
            f"📅 Письмо будет открыто <b>{open_date_local.strftime('%d.%m.%Y в %H:%M')}</b>\n\n"
            "Теперь напиши текст письма:",
            reply_markup=get_cancel_fsm_kb(),
            parse_mode="HTML"
        )
    except ValueError:
        await message.answer(
            "Неверное время. Используй формат <b>ЧЧ:ММ</b> (24 часа)", 
            reply_markup=get_cancel_fsm_kb(),
            parse_mode="HTML"
        )

@router.message(LetterStates.entering_text, F.text)
async def process_letter_text(
    message: Message,
    state: FSMContext,
    user_service: UserService,
) -> None:
    text = message.text.strip()
    
    if not text:
        await message.answer("Напиши что-нибудь ❤️", reply_markup=get_cancel_fsm_kb())
        return
    
    if len(text) > 4096:
        await message.answer(
            f"Слишком длинно. Максимум 4096 символов. Сейчас: {len(text)}",
            reply_markup=get_cancel_fsm_kb()
        )
        return
    
    await state.update_data(text=text)
    await state.set_state(LetterStates.confirming)
    
    data = await state.get_data()
    open_date_str = data.get("open_date")
    
    if not open_date_str:
        await message.answer(
            "Ошибка данных. Начни создание письма заново", 
            reply_markup=get_letter_date_keyboard()
        )
        await state.set_state(LetterStates.choosing_date)
        return
    
    open_date_utc = datetime.fromisoformat(open_date_str)
    
    user = await user_service.get_profile(telegram_id=message.from_user.id)
    user_tz = user.timezone if user else "UTC"
    open_date_local = to_local(open_date_utc, user_tz)
    
    await message.answer(
        f"💌 <b>Подтверждение</b>\n\n"
        f"Дата и время открытия: <b>{open_date_local.strftime('%d.%m.%Y в %H:%M')}</b>\n"
        f"Текст письма:\n\n"
        f"<i>{text}</i>\n\n"
        f"Письмо будет доставлено в указанное время.",
        reply_markup=get_letter_confirm_keyboard(),
        parse_mode="HTML"
    )


@router.message(LetterStates.entering_text, ~F.text)

    )


@router.callback_query(LetterAction.filter(F.action == "confirm"), LetterStates.confirming)
async def cb_confirm_letter(
    callback: CallbackQuery,
    state: FSMContext,
    couple_service: CoupleService,
    letter_service: LetterService,
    user_service: UserService,
) -> None:
    user_id = callback.from_user.id
    data = await state.get_data()
    text = data.get("text")
    open_date_str = data.get("open_date")
    
    if not text or not open_date_str:
        await callback.answer("Ошибка данных. Начни сначала", show_alert=True)
        await state.clear()
        return
    
    open_date_utc = datetime.fromisoformat(open_date_str)
    
    couple = await couple_service.get_couple(user_id)
    if not couple:
        await callback.answer("Ты не в паре", show_alert=True)
        await state.clear()
        return
    
    receiver_id = couple.user2_id if couple.user1_id == user_id else couple.user1_id
    
    await letter_service.create_letter(
        couple_id=couple.id,
        sender_id=user_id,
        receiver_id=receiver_id,
        text=text,
        open_date=open_date_utc
    )
    
    await state.clear()
    
    user = await user_service.get_profile(telegram_id=user_id)
    user_tz = user.timezone if user else "UTC"
    open_date_local = to_local(open_date_utc, user_tz)
    
    await callback.message.edit_text(
        "💌 Письмо запечатано!\n\n"
        f"Оно будет доставлено {open_date_local.strftime('%d.%m.%Y в %H:%M')}.",
        parse_mode="HTML"
    )
    await callback.answer()

@router.message(Command("letters"))
async def cmd_show_letters(
    message: Message,
    letter_service: LetterService,
    couple_service: CoupleService,
    user_service: UserService,
) -> None:
    """Показывает все активные письма в виде красивого списка"""
    user_id = message.from_user.id
    
    couple = await couple_service.get_couple(user_id)
    if not couple:
        await message.answer("Ты не в паре. Сначала создай пару 💑", parse_mode="HTML")
        return
    
    user = await user_service.get_profile(telegram_id=user_id)
    user_tz = user.timezone if user else "UTC"
    
    letters = await letter_service.get_active_letters(user_id)
    
    if not letters:
        await message.answer(
            "У тебя нет активных писем 💌\n\n"
            "Создай первое письмо через меню пары!",
            parse_mode="HTML"
        )
        return
    
    lines = ["💌 <b>Твои активные письма:</b>\n"]
    
    for i, letter in enumerate(letters, 1):
        open_date_local = to_local(letter.open_date, user_tz)
        date_str = open_date_local.strftime('%d.%m.%Y в %H:%M')
        
        preview = letter.text[:100]
        if len(letter.text) > 100:
            preview += "..."
        
        lines.append(
            f"<b>{i}.</b> 📅 {date_str}\n"
            f"<i>{preview}</i>\n"
        )
    
    lines.append("\n<i>Письма нельзя изменить после создания.</i>")
    
    await message.answer("\n".join(lines), parse_mode="HTML")

@router.callback_query(StateFilter(None), MemoryAction.filter(F.action == "list"))
async def cb_show_memories(
    callback: CallbackQuery,
    couple_service: CoupleService,
    memory_service: MemoryService,
) -> None:
    """Показывает список воспоминаний пары."""
    user_id = callback.from_user.id

    couple = await couple_service.get_couple(user_id)

    if not couple:
        await callback.answer("Сначала создай пару 💑", show_alert=True)
        return

    memories = await memory_service.get_memories(couple.id)

    if memories:
        text = (
            "📖 <b>История отношений</b>\n\n"
            "Выбери воспоминание, чтобы посмотреть детали:"
        )
    else:
        text = (
            "📖 <b>История отношений</b>\n\n"
            "Пока нет воспоминаний.\n"
            "Добавьте первое важное событие 💞"
        )

    if callback.message:
        await callback.message.edit_text(
            text,
            reply_markup=get_memories_list_kb(memories),
            parse_mode="HTML",
        )

    await callback.answer()


@router.callback_query(StateFilter(None), MemoryAction.filter(F.action == "view"))
async def cb_view_memory(
    callback: CallbackQuery,
    callback_data: MemoryAction,
    couple_service: CoupleService,
    memory_service: MemoryService,
) -> None:
    """Показывает детали воспоминания."""
    user_id = callback.from_user.id
    memory_id = callback_data.memory_id

    if memory_id is None:
        await callback.answer()
        return

    memory = await memory_service.get_memory(memory_id)

    if not memory:
        await callback.answer("Воспоминание не найдено", show_alert=True)
        return

    couple = await couple_service.get_couple(user_id)

    if not couple or couple.id != memory.couple_id:
        await callback.answer("Нет доступа к этому воспоминанию", show_alert=True)
        return

    memory_date = memory.memory_date.strftime("%d.%m.%Y")

    text = (
        f"📖 <b>{escape(memory.title)}</b>\n\n"
        f"📅 <b>Дата события:</b> {memory_date}\n"
    )

    if memory.description:
        text += f"\n{escape(memory.description)}\n"

    if callback.message:
        await callback.message.edit_text(
            text,
            reply_markup=get_memory_detail_kb(memory.id),
            parse_mode="HTML",
        )

    await callback.answer()

@router.message(MemoryStates.entering_title, F.text)
async def process_memory_title(message: Message, state: FSMContext) -> None:
    """Обрабатывает название воспоминания."""
    if message.text.startswith("/"):
        return

    title = message.text.strip()

    if not title:
        await message.answer(
            "Введи название.",
            reply_markup=get_cancel_fsm_kb(),
            parse_mode="HTML",
        )
        return

    if len(title) > 120:
        await message.answer(
            "Название слишком длинное, максимум 120 символов.",
            reply_markup=get_cancel_fsm_kb(),
            parse_mode="HTML",
        )
        return

    await state.update_data(title=title)
    await state.set_state(MemoryStates.entering_description)

    await message.answer(
        "Шаг 2/3.\n"
        "Введи описание воспоминания.\n\n"
        "Если не хочешь добавлять описание, отправь:\n"
        "<code>-</code>",
        reply_markup=get_cancel_fsm_kb(),
        parse_mode="HTML",
    )

@router.message(MemoryStates.entering_description, F.text)
async def process_memory_description(message: Message, state: FSMContext) -> None:
    """Обрабатывает описание воспоминания."""
    if message.text.startswith("/"):
        return

    description = message.text.strip()

    if description == "-":
        description = None

    await state.update_data(description=description)

@router.message(MemoryStates.entering_date, F.text)
async def process_memory_date(
    message: Message,
    state: FSMContext,
    couple_service: CoupleService,
    memory_service: MemoryService,
) -> None:
    """Обрабатывает дату воспоминания и создаёт его."""
    if message.text.startswith("/"):
        return

    user_id = message.from_user.id

    try:
        memory_date = datetime.strptime(message.text.strip(), "%d.%m.%Y").date()
    except ValueError:
        await message.answer(
            "Неверная дата. Используй формат <b>ДД.ММ.ГГГГ</b>.",
            reply_markup=get_cancel_fsm_kb(),
            parse_mode="HTML",
        )
        return

    if memory_date > datetime.now(timezone.utc).date():
        await message.answer(
            "Дата должна быть в прошлом.",
            reply_markup=get_cancel_fsm_kb(),
            parse_mode="HTML",
        )
        return

    couple = await couple_service.get_couple(user_id)

    if not couple:
        await state.clear()
        await message.answer(
            "Сначала создай пару 💑",
            parse_mode="HTML",
        )
        return

    data = await state.get_data()

    title = data.get("title")
    description = data.get("description")

    if not title:
        await state.clear()
        await message.answer(
            "Ошибка данных. Попробуй добавить воспоминание заново.",
            parse_mode="HTML",
        )
        return

    try:
        await memory_service.create_memory(
            couple_id=couple.id,
            author_id=user_id,
            title=title,
            description=description,
            memory_date=memory_date,
        )
    except ValueError as e:
        await message.answer(
            str(e),
            reply_markup=get_cancel_fsm_kb(),
            parse_mode="HTML",
        )
        return

    await state.clear()

    memories = await memory_service.get_memories(couple.id)

    await message.answer(
        "✅ Воспоминание добавлено!",
        parse_mode="HTML",
    )

    await message.answer(
        "📖 <b>История отношений</b>\n\n"
        "Выбери воспоминание, чтобы посмотреть детали:",
        reply_markup=get_memories_list_kb(memories),
        parse_mode="HTML",
    )

@router.message(Command("profile"))
async def cmd_profile(
    message: Message,
    user_service: UserService,
    couple_service: CoupleService,
) -> None:
    """Показывает профиль пользователя."""
    user_id = message.from_user.id
    user = await user_service.get_profile(telegram_id=user_id)
    
    if not user:
        await message.answer("❌ Профиль не найден. Пожалуйста, отправь /start для регистрации.")
        return

    couple = await couple_service.get_couple(user_id=user_id)

    notifications_status = "✅ Включены" if user.notifications else "❌ Выключены"
    
    # Конвертируем дату регистрации из UTC в локальное время пользователя
    local_created_at = to_local(user.created_at, user.timezone)
    
    text = (
        f"👤 <b>Твой профиль</b>\n\n"
        f"<b>Имя:</b> {user.first_name}\n"
        f"<b>Username:</b> {user.username or 'не указан'}\n"
        f"<b>Часовой пояс:</b> 🌍 {user.timezone}\n"
        f"<b>Уведомления:</b> {notifications_status}\n\n"
        f"<b>Дата регистрации:</b> {local_created_at.strftime('%d.%m.%Y')}"
    )

    if couple:
        partner_id = couple.user2_id if couple.user1_id == user_id else couple.user1_id
        partner = await user_service.get_profile(partner_id)
        partner_name = (partner.first_name or partner.username or "Неизвестно") if partner else "Неизвестно"
        
        # Конвертируем дату активации пары в локальное время пользователя
        local_activated_at = to_local(couple.activated_at, user.timezone) if couple.activated_at else None
        date_str = local_activated_at.strftime("%d.%m.%Y") if local_activated_at else "Неизвестно"

        text += (
            f"\n\n💑 <b>В паре с:</b> {partner_name}\n"
            f" <b>Вместе с:</b> {date_str}"
        )
    else:
        text += "\n\n💔 <b>Статус:</b> Не в паре"

    # Создаем inline-клавиатуру с кнопкой "Назад"
    back_kb = InlineKeyboardMarkup(
        inline_keyboard=[
            [InlineKeyboardButton(text="🔙 Назад", callback_data=NavAction(action="back").pack())]
        ]
    )

    await message.answer(text, parse_mode="HTML", reply_markup=back_kb)

from typing import Any, Awaitable, Callable, Dict

from aiogram import BaseMiddleware
from aiogram.types import TelegramObject
from sqlalchemy.ext.asyncio import async_sessionmaker

from database.repositories.user import UserRepository
from database.repositories.couple import CoupleRepository
from database.repositories.letter import LetterRepository
from database.repositories.memory import MemoryRepository

from services.user_service import UserService
from services.couple_service import CoupleService
from services.letter_service import LetterService
from services.memory_service import MemoryService


class DbSessionMiddleware(BaseMiddleware):
    """Middleware для управления сессиями БД и проброса сервисов в хэндлеры."""

    def __init__(self, session_pool: async_sessionmaker) -> None:
        super().__init__()
        self.session_pool = session_pool

    async def __call__(
        self,
        handler: Callable[[TelegramObject, Dict[str, Any]], Awaitable[Any]],
        event: TelegramObject,
        data: Dict[str, Any],
    ) -> Any:
        async with self.session_pool() as session:
            # Репозитории
            user_repo = UserRepository(session)
            couple_repo = CoupleRepository(session)
            letter_repo = LetterRepository(session)
            memory_repo = MemoryRepository(session)

            # Bot уже находится в data благодаря aiogram
            bot = data["bot"]

            # Сервисы
            user_service = UserService(user_repo)
            letter_service = LetterService(letter_repo, user_repo, bot)
            couple_service = CoupleService(couple_repo, letter_service, bot)
            memory_service = MemoryService(memory_repo, couple_repo)

            # Пробрасываем зависимости в хэндлеры
            data["session"] = session
            data["user_repo"] = user_repo
            data["user_service"] = user_service
            data["couple_repo"] = couple_repo
            data["couple_service"] = couple_service
            data["letter_service"] = letter_service
            data["memory_repo"] = memory_repo
            data["memory_service"] = memory_service

            try:
                result = await handler(event, data)
                await session.commit()
                return result
            except Exception:
                await session.rollback()
                raise

async def back_to_main_menu(target: Message | CallbackQuery, state: FSMContext) -> None:

                chat_id=chat_id,
                text="🏠 Главное меню",
                reply_markup=get_main_menu_kb()
            )

        await target.answer(
            text="🏠 Главное меню",
            reply_markup=get_main_menu_kb()

from datetime import datetime, timezone
from zoneinfo import ZoneInfo, ZoneInfoNotFoundError
import logging

logger = logging.getLogger(__name__)


def get_common_timezones() -> list[str]:
    """Возвращает список популярных часовых поясов для выбора пользователем."""
    return [
        "Europe/Moscow",
        "Europe/Kyiv",
        "Europe/Minsk",
        "Europe/Berlin",
        "Europe/London",
        "America/New_York",
        "Asia/Almaty",
        "Asia/Tashkent",
        "Asia/Dubai",
        "UTC",
    ]


def validate_timezone(tz_name: str) -> bool:
    """
    Проверяет, является ли строка валидным названием часового пояса.
    Возвращает True, если пояс существует, иначе False.
    """
    try:
        ZoneInfo(tz_name)
        return True
    except ZoneInfoNotFoundError:
        logger.warning(f"Невалидный часовой пояс: {tz_name}")
        return False
    except Exception as e:
        logger.error(f"Ошибка при проверке часового пояса {tz_name}: {e}")
        return False


def to_utc(dt: datetime, tz_name: str) -> datetime:
    """
    Конвертирует наивное (без часового пояса) или локальное datetime 
    в UTC для сохранения в базу данных.
    
    :param dt: Объект datetime (предполагается, что он в локальном времени пользователя)
    :param tz_name: Строка с названием часового пояса пользователя (например, "Europe/Moscow")
    :return: Объект datetime с часовым поясом UTC
    """
    if not validate_timezone(tz_name):
        tz_name = "UTC" # Fallback на случай невалидного пояса
    
    target_tz = ZoneInfo(tz_name)
    
    # Если datetime "наивный" (без tzinfo), считаем, что он в локальном поясе пользователя
    if dt.tzinfo is None:
        dt = dt.replace(tzinfo=target_tz)
    
    # Конвертируем в UTC
    return dt.astimezone(timezone.utc)


def to_local(dt: datetime, tz_name: str) -> datetime:
    """
    Конвертирует datetime из UTC (например, из БД) в локальное время пользователя для отображения.
    
    :param dt: Объект datetime (предполагается, что он в UTC)
    :param tz_name: Строка с названием часового пояса пользователя
    :return: Объект datetime в локальном часовом поясе
    """
    if not validate_timezone(tz_name):
        tz_name = "UTC"
    
    target_tz = ZoneInfo(tz_name)
    
    # Если datetime "наивный", считаем, что это UTC (стандарт для БД)
    if dt.tzinfo is None:
        dt = dt.replace(tzinfo=timezone.utc)
    
    # Конвертируем в локальный пояс
    return dt.astimezone(target_tz)

from typing import Generic, Optional, Sequence, Type, TypeVar

from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from database.models import Base

ModelType = TypeVar("ModelType", bound=Base)


class BaseRepository(Generic[ModelType]):
    """Базовый асинхронный репозиторий для выполнения CRUD-операций."""

    def __init__(self, model: Type[ModelType], session: AsyncSession) -> None:
        self.model = model
        self.session = session

    async def get_by_id(self, id: int) -> Optional[ModelType]:
        """Получить запись по primary key (id)."""
        return await self.session.get(self.model, id)

    async def get_all(self) -> Sequence[ModelType]:
        """Получить все записи сущности."""
        stmt = select(self.model)
        result = await self.session.execute(stmt)
        return result.scalars().all()

    async def add(self, entity: ModelType) -> ModelType:
        """Добавить новую сущность в сессию."""
        self.session.add(entity)
        await self.session.flush()
        return entity

    async def delete(self, entity: ModelType) -> None:
        """Удалить сущность из сессии."""
        await self.session.delete(entity)
        await self.session.flush()

from datetime import datetime

from sqlalchemy import select, update

from database.models.letter import Letter
from database.repositories.base import BaseRepository


class LetterRepository(BaseRepository[Letter]):
    
    def __init__(self, session) -> None:
        super().__init__(Letter, session)

    async def create(self, letter: Letter) -> Letter:
        """Создаёт письмо и возвращает его с проставленным id."""
        self.session.add(letter)
        await self.session.flush()
        return letter

    async def get_pending(self, now: datetime) -> list[Letter]:
        """Находит все письма, которые пора доставить.
        
        Использует FOR UPDATE SKIP LOCKED для предотвращения гонки:
        если два воркера запустятся одновременно, каждый получит свой набор писем.
        """
        query = (
            select(Letter)
            .where(
                Letter.status == "pending",
                Letter.open_date <= now
            )
            .with_for_update(skip_locked=True)  # Блокируем строки, другие транзакции их пропустят
        )
        result = await self.session.execute(query)
        return list(result.scalars().all())

    async def mark_delivered(self, letter_id: int, delivered_at: datetime) -> None:
        """Помечает письмо как доставленное."""
        query = (
            update(Letter)
            .where(Letter.id == letter_id)
            .values(status="delivered", delivered_at=delivered_at)
        )
        await self.session.execute(query)
        await self.session.flush()

    async def cancel_by_couple(self, couple_id: int) -> None:
        """Отменяет все активные письма пары при разрыве."""
        query = (
            update(Letter)
            .where(Letter.couple_id == couple_id, Letter.status == "pending")
            .values(status="cancelled")
        )

from datetime import date

from sqlalchemy import delete as sa_delete
from sqlalchemy import extract, func, or_, select, update
from sqlalchemy.ext.asyncio import AsyncSession

from database.models.couple import Couple
from database.models.memory import Memory
from database.repositories.base import BaseRepository


class MemoryRepository(BaseRepository[Memory]):
    def __init__(self, session: AsyncSession) -> None:
        super().__init__(Memory, session)

    async def create(self, memory: Memory) -> Memory:
        """Создаёт воспоминание."""
        self.session.add(memory)
        await self.session.flush()
        return memory

    async def get_by_couple(self, couple_id: int) -> list[Memory]:
        """Возвращает воспоминания пары, отсортированные по дате события."""
        stmt = (
            select(Memory)
            .where(Memory.couple_id == couple_id)
            .order_by(Memory.memory_date.desc(), Memory.id.desc())
        )
        result = await self.session.execute(stmt)
        return list(result.scalars().all())

    async def delete_by_id(self, memory_id: int) -> None:
        """Удаляет воспоминание по ID."""
        stmt = sa_delete(Memory).where(Memory.id == memory_id)
        await self.session.execute(stmt)
        await self.session.flush()

    async def exists_by_title_and_date(
        self,
        couple_id: int,
        title: str,
        memory_date: date,
    ) -> bool:
        """Проверяет, есть ли уже у пары такое воспоминание."""
        stmt = (
            select(func.count())
            .select_from(Memory)
            .where(
                Memory.couple_id == couple_id,
                func.lower(Memory.title) == title.lower(),
                Memory.memory_date == memory_date,
            )
        )
        result = await self.session.execute(stmt)
        return result.scalar_one() > 0

    async def get_anniversaries_today(self, today: date) -> list[Memory]:
        """
        Возвращает годовщины, которые нужно напомнить сегодня.

        Условия:
        - месяц и день совпадают с сегодняшним днём;
        - год события меньше текущего года;
        - напоминание ещё не отправлялось сегодня;
        - пара активна.
        """
        stmt = (
            select(Memory)
            .join(Couple, Memory.couple_id == Couple.id)
            .where(
                Couple.status == "active",
                extract("month", Memory.memory_date) == today.month,
                extract("day", Memory.memory_date) == today.day,
                extract("year", Memory.memory_date) < today.year,
                or_(
                    Memory.last_notified.is_(None),
                    Memory.last_notified != today,
                ),
            )
            .order_by(Memory.memory_date.asc())
        )
        result = await self.session.execute(stmt)
        return list(result.scalars().all())

    async def mark_notified(self, memory_id: int, notified_date: date) -> None:
        """Отмечает, что напоминание за эту дату уже было отправлено."""
        stmt = (
            update(Memory)
            .where(Memory.id == memory_id)
            .values(last_notified=notified_date)
        )
        await self.session.execute(stmt)
        await self.session.flush()

from typing import Optional

from sqlalchemy import select
from sqlalchemy.dialects.postgresql import insert as pg_insert
from sqlalchemy.ext.asyncio import AsyncSession

from database.models.user import User
from database.repositories.base import BaseRepository


class UserRepository(BaseRepository[User]):
    """Репозиторий для работы с таблицей пользователей."""

    def __init__(self, session: AsyncSession) -> None:
        super().__init__(User, session)

    async def get_by_tg_id(self, telegram_id: int) -> Optional[User]:
        """Получить пользователя по Telegram ID."""
        stmt = select(User).where(User.telegram_id == telegram_id)
        result = await self.session.execute(stmt)
        return result.scalar_one_or_none()
    
    async def get_by_id(self, user_id: int) -> Optional[User]:
        """Получить пользователя по внутреннему ID."""
        stmt = select(User).where(User.id == user_id)
        result = await self.session.execute(stmt)
        return result.scalar_one_or_none()

    async def upsert(
        self,
        telegram_id: int,
        first_name: str,
        username: Optional[str] = None,
        last_name: Optional[str] = None,
    ) -> User:
        """
        Атомарный UPSERT через PostgreSQL ON CONFLICT.
        Если пользователя нет — создает. Если есть — обновляет имя.
        """
        stmt = (
            pg_insert(User)
            .values(
                telegram_id=telegram_id,
                username=username,
                first_name=first_name,
                last_name=last_name,
            )
            .on_conflict_do_update(
                index_elements=[User.telegram_id],
                set_={
                    "username": username,
                    "first_name": first_name,
                    "last_name": last_name,
                },
            )
            .returning(User)
        )

        result = await self.session.execute(stmt)
        return result.scalar_one()

    async def update_settings(
        self,
        telegram_id: int,
        timezone: Optional[str] = None,
        notifications: Optional[bool] = None,
    ) -> Optional[User]:
        """
        Обновляет только переданные настройки.
        Изменения зафиксирует DbSessionMiddleware при коммите.
        """
        user = await self.get_by_tg_id(telegram_id)
        if not user:
            return None

        if timezone is not None:
            user.timezone = timezone
        if notifications is not None:
            user.notifications = notifications

        return user

import secrets
import string
from datetime import datetime, timedelta, timezone

from sqlalchemy.exc import IntegrityError
from aiogram import Bot
from aiogram.exceptions import TelegramBadRequest

from database.repositories.couple import CoupleRepository
from database.models.couple import Couple
from services.letter_service import LetterService


class CoupleService:
    def __init__(
        self, 
        couple_repo: CoupleRepository,
        letter_service: LetterService,
        bot: Bot
    ):
        self.couple_repo = couple_repo
        self.letter_service = letter_service
        self.bot = bot

    @staticmethod
    def _generate_invite_code() -> str:
        """Генерация 6-символьного кода без неоднозначных символов"""
        alphabet = string.ascii_uppercase.replace("O", "").replace("I", "").replace("L", "")
        alphabet += string.digits.replace("0", "").replace("1", "")
        return "".join(secrets.choice(alphabet) for _ in range(6))

    async def create_couple(self, user_id: int) -> Couple:
        """Создать новую пару или обновить код в pending-паре"""
        existing = await self.couple_repo.get_by_user_id(user_id)
        if existing:
            raise ValueError("Ты уже в паре")

        invite_code = self._generate_invite_code()
        invite_expires = datetime.now(timezone.utc) + timedelta(hours=24)

        pending = await self.couple_repo.get_pending_by_user_id(user_id)

        if pending:
            try:
                return await self.couple_repo.update_invite_code(
                    couple_id=pending.id,
                    invite_code=invite_code,
                    invite_expires=invite_expires,
                )
            except IntegrityError:
                raise ValueError("Не удалось создать код. Попробуй ещё раз")

        try:
            return await self.couple_repo.create_couple(
                user1_id=user_id,
                invite_code=invite_code,
                invite_expires=invite_expires,
            )
        except IntegrityError:
            raise ValueError("Не удалось создать код. Попробуй ещё раз")

    async def join_couple(self, user_id: int, invite_code: str) -> Couple:
        """Присоединиться к паре по коду"""
        existing = await self.couple_repo.get_by_user_id(user_id)
        if existing:
            raise ValueError("Ты уже в паре")

        couple = await self.couple_repo.get_by_invite_code(invite_code)
        if not couple:
            raise ValueError("Код не найден")

        if couple.invite_expires < datetime.now(timezone.utc):
            raise ValueError("Код истёк")

        if couple.user1_id == user_id:
            raise ValueError("Нельзя присоединиться к самому себе")

        activated = await self.couple_repo.activate_couple(couple.id, user_id)

        # Если пользователь до этого создавал свою pending-пару — отменяем её
        await self.couple_repo.cancel_pending_by_user(user_id)

        return activated

    async def get_couple(self, user_id: int) -> Couple | None:
        """Получить активную пару пользователя"""
        return await self.couple_repo.get_by_user_id(user_id)

    async def leave_couple(self, user_id: int) -> None:
        """Выйти из пары"""
        couple = await self.couple_repo.get_by_user_id(user_id)
        if not couple:
            raise ValueError("Ты не в паре")

        # 1. Разрываем пару
        await self.couple_repo.break_couple(couple.id)
        
        # 2. Отменяем все активные письма (ВАЖНО!)
        await self.letter_service.cancel_pending_letters(couple.id)
        
        # 3. Уведомляем партнера
        partner_id = couple.user2_id if couple.user1_id == user_id else couple.user1_id
        if partner_id:
            try:
                await self.bot.send_message(
                    chat_id=partner_id,
                    text="💔 Твой партнер решил завершить отношения. Пара распалась."
                )
            except TelegramBadRequest:
                pass  # Партнер мог заблокировать бота

import logging

from aiogram import Bot
from aiogram.exceptions import TelegramBadRequest

from bot.utils.timezone_utils import to_local
from database.models.letter import Letter
from database.repositories.letter import LetterRepository

logger = logging.getLogger(__name__)


class LetterService:
    def __init__(
        self, 
        letter_repo: LetterRepository, 
        user_repo: UserRepository, 
        bot: Bot
    ) -> None:
        self._letter_repo = letter_repo
        self._user_repo = user_repo
        self._bot = bot

    async def create_letter(
        self, 
        couple_id: int, 
        sender_id: int, 
        receiver_id: int, 
        text: str, 
        open_date: datetime
    ) -> Letter:
        """Создаёт новое письмо."""
        letter = Letter(
            couple_id=couple_id,
            sender_id=sender_id,
            receiver_id=receiver_id,
            text=text,
            open_date=open_date,
            status="pending"
        )
        return await self._letter_repo.create(letter)

    async def get_pending_letters(self, now: datetime) -> list[Letter]:
        """Возвращает все письма, которые пора доставить."""
        return await self._letter_repo.get_pending(now)

    async def get_letter_by_id(self, letter_id: int) -> Letter | None:
        """Получает письмо по ID."""
        return await self._letter_repo.get_by_id(letter_id)

    async def deliver_letter(self, letter: Letter) -> None:
        """Отправляет письмо, обновляет БД и уведомляет отправителя."""
        sender = await self._user_repo.get_by_tg_id(letter.sender_id)
        receiver = await self._user_repo.get_by_tg_id(letter.receiver_id)
        
        sender_name = (sender.first_name or sender.username or "Твой партнер") if sender else "Твой партнер"
        receiver_name = (receiver.first_name or receiver.username or "партнер") if receiver else "партнер"

        receiver_tz = receiver.timezone if receiver else "UTC"
        sender_tz = sender.timezone if sender else "UTC"

        # 1. Отправляем письмо получателю С КНОПКАМИ
        try:
            created_local = to_local(letter.created_at, receiver_tz)
            opened_local = to_local(letter.open_date, receiver_tz)
            
            # Импортируем клавиатуру здесь, чтобы избежать циклических импортов
            from bot.keyboards.letter_delivery_kb import get_letter_delivery_kb
            
            await self._bot.send_message(
                chat_id=letter.receiver_id,
                text=(
                    f"💌 <b>Письмо из капсулы времени!</b>\n\n"
                    f"От: {sender_name}\n"
                    f"Дата написания: {created_local.strftime('%d.%m.%Y')}\n"
                    f"Дата открытия: {opened_local.strftime('%d.%m.%Y в %H:%M')}\n\n"
                    f"✨ <i>{letter.text}</i>"
                ),
                reply_markup=get_letter_delivery_kb(letter.id),
                parse_mode="HTML"
            )
        except TelegramBadRequest as e:
            logger.warning(f"Не удалось доставить письмо {letter.id} пользователю {letter.receiver_id}: {e}")

        # 2. Помечаем как доставленное в БД
        await self._letter_repo.mark_delivered(letter.id, datetime.now(timezone.utc))

        # 3. Уведомляем отправителя
        try:
            delivered_local = to_local(datetime.now(timezone.utc), sender_tz)
            await self._bot.send_message(
                chat_id=letter.sender_id,
                text=(
                    f"💌 Твоё письмо для {receiver_name} было доставлено!\n"
                    f"📅 {delivered_local.strftime('%d.%m.%Y в %H:%M')}"
                )
            )
        except TelegramBadRequest as e:
            logger.warning(f"Не удалось уведомить отправителя {letter.sender_id} о письме {letter.id}: {e}")

    async def cancel_pending_letters(self, couple_id: int) -> None:
        """Отменяет все активные письма пары (при разрыве)."""
        await self._letter_repo.cancel_by_couple(couple_id)

    async def get_active_letters(self, user_id: int) -> list[Letter]:
        """Возвращает все ожидающие письма пользователя."""
        return await self._letter_repo.get_active_by_user(user_id)

from datetime import date, datetime, timezone

from database.models.memory import Memory
from database.repositories.couple import CoupleRepository
from database.repositories.memory import MemoryRepository


class MemoryService:
    def __init__(
        self,
        memory_repo: MemoryRepository,
        couple_repo: CoupleRepository,
    ) -> None:
        self._memory_repo = memory_repo
        self._couple_repo = couple_repo

    async def create_memory(
        self,
        couple_id: int,
        author_id: int,
        title: str,
        description: str | None,
        memory_date: date,
    ) -> Memory:
        """Создаёт воспоминание после базовой валидации."""
        title = title.strip()

        if not title:
            raise ValueError("Введи название")

        if len(title) > 120:
            raise ValueError("Название слишком длинное, максимум 120 символов")

        if description is not None:
            description = description.strip()
            if not description:
                description = None

        today = datetime.now(timezone.utc).date()

        if memory_date > today:
            raise ValueError("Дата должна быть в прошлом")

        is_duplicate = await self._memory_repo.exists_by_title_and_date(
            couple_id=couple_id,
            title=title,
            memory_date=memory_date,
        )

        if is_duplicate:
            raise ValueError("Такое воспоминание уже добавлено")

        memory = Memory(
            couple_id=couple_id,
            author_id=author_id,
            title=title,
            description=description,
            memory_date=memory_date,
        )

        return await self._memory_repo.create(memory)

    async def get_memories(self, couple_id: int) -> list[Memory]:
        """Возвращает все воспоминания пары."""
        return await self._memory_repo.get_by_couple(couple_id)

    async def get_memory(self, memory_id: int) -> Memory | None:
        """Возвращает воспоминание по ID."""
        return await self._memory_repo.get_by_id(memory_id)

    async def delete_memory(self, memory_id: int, user_id: int) -> None:
        """Удаляет воспоминание, если пользователь состоит в этой паре."""
        memory = await self._memory_repo.get_by_id(memory_id)

        if not memory:
            raise ValueError("Воспоминание не найдено")

        couple = await self._couple_repo.get_by_user_id(user_id)

        if not couple or couple.id != memory.couple_id:
            raise ValueError("Нет доступа к этому воспоминанию")

        await self._memory_repo.delete_by_id(memory_id)
